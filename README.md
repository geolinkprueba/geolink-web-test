        // Lógica del Speedtest Interactivo
        function iniciarTestVelocidad() {
            const btn = document.getElementById('start-test-btn');
            const display = document.getElementById('speed-display');
            const status = document.getElementById('speed-status');
            const pingEl = document.getElementById('metric-ping');
            const downloadEl = document.getElementById('metric-download');
            const uploadEl = document.getElementById('metric-upload');

            btn.disabled = true;
            btn.classList.add('opacity-50', 'cursor-not-allowed');
            
            status.innerText = "Midiendo latencia (Ping a Nodo Girardota)...";
            pingEl.innerText = "12 ms";
            
            let velocidadActual = 0;
            let subidaActual = 0;
            
            setTimeout(() => {
                status.innerText = "Probando velocidad de descarga...";
                let downloadInterval = setInterval(() => {
                    velocidadActual += 4.5;
                    if (velocidadActual >= 48.2) {
                        velocidadActual = 48.2;
                        clearInterval(downloadInterval);
                        downloadEl.innerText = velocidadActual.toFixed(1) + " Mbps";
                        
                        status.innerText = "Probando velocidad de subida...";
                        let uploadInterval = setInterval(() => {
                            subidaActual += 1.8;
                            if (subidaActual >= 18.5) {
                                subidaActual = 18.5;
                                clearInterval(uploadInterval);
                                uploadEl.innerText = subidaActual.toFixed(1) + " Mbps";
                                status.innerText = "¡Prueba completada con éxito!";
                                display.innerHTML = `48.2 <span class="text-2xl text-[#00f2fe] font-semibold">Mbps</span>`;
                                
                                btn.disabled = false;
                                btn.classList.remove('opacity-50', 'cursor-not-allowed');
                                btn.innerText = "Repetir Test de Velocidad";
                            }
                            display.innerHTML = `${subidaActual.toFixed(1)} <span class="text-2xl text-amber-400 font-semibold">Mbps</span>`;
                        }, 50);

                    }
                    display.innerHTML = `${velocidadActual.toFixed(1)} <span class="text-2xl text-[#00f2fe] font-semibold">Mbps</span>`;
                }, 40);
            }, 1000);
        }
    </script>
